# 异步数据采集系统详细开发需求文档

## 1. 项目概述

### 1.1 项目背景
开发一个高性能、可扩展的异步数据采集系统，用于从六大主流广告平台（百度、搜狗、360、神马、字节跳动、腾讯）获取商业广告数据，实现数据的统一采集、处理、存储和监控。

### 1.2 设计原则
- **模块化**：各功能模块独立设计，低耦合高内聚
- **通用性**：支持多种广告平台的统一接入
- **可复用**：核心组件可在其他项目中复用
- **可扩展**：支持新平台接入和功能扩展
- **稳定性**：完善的错误处理和恢复机制
- **可靠性**：数据一致性保证和监控告警

### 1.3 技术栈
- **语言**：Python 3.11+
- **异步框架**：asyncio + aiohttp
- **数据库**：MySQL 8.0+ with SQLAlchemy 2.0
- **数据验证**：Pydantic 2.0
- **容器化**：Docker + Docker Compose
- **监控**：自定义日志系统 + 指标监控

from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
from sqlalchemy.orm import sessionmaker
from contextlib import asynccontextmanager

class DatabaseManager:
    def __init__(self, config: DatabaseConfig):
        self.config = config
        self.engine = None
        self.session_factory = None
    
    async def initialize(self):
        """初始化数据库连接"""
        self.engine = create_async_engine(
            self._build_url(),
            pool_size=self.config.pool_size,
            max_overflow=self.config.max_overflow,
            pool_timeout=self.config.pool_timeout,
            echo=self.config.debug
        )
        self.session_factory = sessionmaker(
            self.engine, class_=AsyncSession, expire_on_commit=False
        )
    
    @asynccontextmanager
    async def get_session(self):
        """获取数据库会话"""
        async with self.session_factory() as session:
            try:
                yield session
                await session.commit()
            except Exception:
                await session.rollback()
                raise
    
    async def health_check(self) -> bool:
        """数据库健康检查"""
        try:
            async with self.get_session() as session:
                await session.execute("SELECT 1")
            return True
        except Exception:
            return False
```

import aiohttp
from typing import Optional, Dict, Any
from utils.retry import retry_async

class HTTPClient:
    def __init__(self, config: HTTPConfig):
        self.config = config
        self.session: Optional[aiohttp.ClientSession] = None
    
    async def __aenter__(self):
        connector = aiohttp.TCPConnector(
            limit=self.config.max_connections,
            limit_per_host=30
        )
        timeout = aiohttp.ClientTimeout(total=self.config.timeout)
        self.session = aiohttp.ClientSession(
            connector=connector,
            timeout=timeout
        )
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if self.session:
            await self.session.close()
    
    @retry_async(max_attempts=3, delay=1.0)
    async def request(
        self,
        method: str,
        url: str,
        headers: Optional[Dict[str, str]] = None,
        params: Optional[Dict[str, Any]] = None,
        json_data: Optional[Dict[str, Any]] = None
    ) -> Dict[str, Any]:
        """发送HTTP请求"""
        async with self.session.request(
            method, url, headers=headers, params=params, json=json_data
        ) as response:
            response.raise_for_status()
            return await response.json()
```

#### 设计规范
```python
from abc import ABC, abstractmethod
from typing import Dict, Any, List
from models.schemas import AdReportModel

class BasePlatformAdapter(ABC):
    def __init__(self, config: PlatformConfig, http_client: HTTPClient):
        self.config = config
        self.http_client = http_client
        self.access_token: Optional[str] = None
        self.token_expires_at: Optional[datetime] = None
    
    @abstractmethod
    async def authenticate(self) -> str:
        """平台认证，返回访问令牌"""
        pass
    
    @abstractmethod
    async def fetch_report(
        self, 
        start_date: date, 
        end_date: date, 
        **kwargs
    ) -> List[Dict[str, Any]]:
        """获取报表数据"""
        pass
    
    @abstractmethod
    def transform_data(self, raw_data: Dict[str, Any]) -> AdReportModel:
        """数据格式转换"""
        pass
    
    async def ensure_authenticated(self):
        """确保认证有效"""
        if not self.access_token or self._is_token_expired():
            self.access_token = await self.authenticate()
    
    def _is_token_expired(self) -> bool:
        """检查令牌是否过期"""
        if not self.token_expires_at:
            return True
        return datetime.now() >= self.token_expires_at
    
    async def get_platform_data(
        self, 
        start_date: date, 
        end_date: date
    ) -> List[AdReportModel]:
        """获取并转换平台数据"""
        await self.ensure_authenticated()
        raw_data = await self.fetch_report(start_date, end_date)
        return [self.transform_data(item) for item in raw_data]
```

### 3.5 数据处理服务 (services/data_processor.py)

#### 功能描述
- 数据清洗和验证
- 批量数据处理
- 数据去重和合并

#### 设计规范
```python
from typing import List, Dict, Any
from models.schemas import AdReportModel
from models.database import AdData
from core.database import DatabaseManager

class DataProcessor:
    def __init__(self, db_manager: DatabaseManager):
        self.db_manager = db_manager
    
    async def process_batch(
        self, 
        platform: str, 
        data_list: List[AdReportModel]
    ) -> Dict[str, int]:
        """批量处理数据"""
        stats = {"processed": 0, "skipped": 0, "errors": 0}
        
        validated_data = []
        for item in data_list:
            try:
                # 数据验证
                validated_item = self._validate_data(item)
                if validated_item:
                    validated_data.append(validated_item)
                    stats["processed"] += 1
                else:
                    stats["skipped"] += 1
            except Exception as e:
                logger.error(f"数据验证失败: {e}")
                stats["errors"] += 1
        
        # 批量入库
        if validated_data:
            await self._batch_insert(platform, validated_data)
        
        return stats
    
    def _validate_data(self, data: AdReportModel) -> Optional[AdReportModel]:
        """数据验证和清洗"""
        # 检查必填字段
        if not data.campaign_id or data.impressions < 0:
            return None
        
        # 数据清洗
        data.cost = round(data.cost, 2)
        return data
    
    async def _batch_insert(
        self, 
        platform: str, 
        data_list: List[AdReportModel]
    ):
        """批量插入数据库"""
        async with self.db_manager.get_session() as session:
            db_objects = [
                AdData(
                    platform=platform,
                    campaign_id=item.campaign_id,
                    metrics=item.dict(exclude={"platform", "campaign_id", "date"}),
                    report_date=item.date
                )
                for item in data_list
            ]
            session.add_all(db_objects)
```

### 3.6 任务调度服务 (services/scheduler.py)

#### 功能描述
- 定时任务管理
- 并发控制
- 任务状态监控

#### 设计规范
```python
import asyncio
from typing import List, Dict, Any
from datetime import datetime, timedelta
from adapters.base import BasePlatformAdapter

class TaskScheduler:
    def __init__(
        self, 
        adapters: Dict[str, BasePlatformAdapter],
        data_processor: DataProcessor,
        max_concurrent_tasks: int = 6
    ):
        self.adapters = adapters
        self.data_processor = data_processor
        self.semaphore = asyncio.Semaphore(max_concurrent_tasks)
        self.running_tasks: Dict[str, asyncio.Task] = {}
    
    async def schedule_daily_collection(self, target_date: date = None):
        """调度每日数据采集"""
        if target_date is None:
            target_date = datetime.now().date() - timedelta(days=1)
        
        tasks = []
        for platform_name, adapter in self.adapters.items():
            task = asyncio.create_task(
                self._collect_platform_data(platform_name, adapter, target_date)
            )
            tasks.append(task)
            self.running_tasks[platform_name] = task
        
        results = await asyncio.gather(*tasks, return_exceptions=True)
        return self._process_results(results)
    
    async def _collect_platform_data(
        self, 
        platform_name: str, 
        adapter: BasePlatformAdapter, 
        target_date: date
    ):
        """采集单个平台数据"""
        async with self.semaphore:
            try:
                logger.info(f"开始采集{platform_name}数据: {target_date}")
                
                # 获取数据
                data_list = await adapter.get_platform_data(
                    target_date, target_date
                )
                
                # 处理数据
                stats = await self.data_processor.process_batch(
                    platform_name, data_list
                )
                
                logger.info(f"{platform_name}数据采集完成: {stats}")
                return {"platform": platform_name, "stats": stats, "success": True}
                
            except Exception as e:
                logger.error(f"{platform_name}数据采集失败: {e}")
                return {"platform": platform_name, "error": str(e), "success": False}
            finally:
                self.running_tasks.pop(platform_name, None)
```

### 5.2 重试机制 (utils/retry.py)

```python
import asyncio
import functools
from typing import Callable, Any
from core.exceptions import PlatformAPIException

def retry_async(
    max_attempts: int = 3,
    delay: float = 1.0,
    backoff: float = 2.0,
    exceptions: tuple = (Exception,)
):
    """异步重试装饰器"""
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        async def wrapper(*args, **kwargs) -> Any:
            last_exception = None
            
            for attempt in range(max_attempts):
                try:
                    return await func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    if attempt == max_attempts - 1:
                        break
                    
                    wait_time = delay * (backoff ** attempt)
                    logger.warning(
                        f"第{attempt + 1}次尝试失败，{wait_time}秒后重试: {e}"
                    )
                    await asyncio.sleep(wait_time)
            
            raise last_exception
        return wrapper
    return decorator
```
